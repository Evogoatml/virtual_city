# balance_checker.py
import requests
import csv
import time
from datetime import datetime

def check_balances(csv_file, output_file="balances_report.csv", max_checks=1000):
    """Check Bitcoin balances for extracted addresses"""

    print(f"🔍 Checking balances for addresses in {csv_file}")
    print(f"📊 Will check maximum {max_checks} addresses")

    balances_found = []
    total_checked = 0
    addresses_with_balance = 0

    try:
        with open(csv_file, 'r') as f:
            reader = csv.DictReader(f)

            with open(output_file, 'w', newline='') as outfile:
                fieldnames = ['public_key', 'private_key', 'address', 'balance_btc', 'balance_usd', 'status']
                writer = csv.DictWriter(outfile, fieldnames=fieldnames)
                writer.writeheader()

                for row in reader:
                    if total_checked >= max_checks:
                        break

                    address = row.get('Derived Address', row.get('address', ''))
                    if not address or address == 'N/A' or address == 'Derivation failed':
                        continue

                    try:
                        # Use blockchain.info API (free tier allows ~1000 requests)
                        url = f"https://blockchain.info/rawaddr/{address}"
                        response = requests.get(url, timeout=10)

                        if response.status_code == 200:
                            data = response.json()
                            balance_satoshis = data.get('final_balance', 0)
                            balance_btc = balance_satoshis / 100000000

                            if balance_btc > 0:
                                addresses_with_balance += 1
                                print(f"💰 FOUND BALANCE! {address}: {balance_btc} BTC")

                                # Get USD value (optional)
                                try:
                                    price_url = "https://api.blockchain.info/ticker"
                                    price_response = requests.get(price_url)
                                    btc_price = price_response.json()['USD']['last']
                                    balance_usd = balance_btc * btc_price
                                except:
                                    balance_usd = 0

                                balances_found.append({
                                    'address': address,
                                    'balance_btc': balance_btc,
                                    'balance_usd': balance_usd
                                })

                            writer.writerow({
                                'public_key': row.get('Public Key', ''),
                                'private_key': row.get('Private Key (WIF)', ''),
                                'address': address,
                                'balance_btc': balance_btc,
                                'balance_usd': balance_usd if 'balance_usd' in locals() else 0,
                                'status': 'Success'
                            })

                        else:
                            writer.writerow({
                                'public_key': row.get('Public Key', ''),
                                'private_key': row.get('Private Key (WIF)', ''),
                                'address': address,
                                'balance_btc': 0,
                                'balance_usd': 0,
                                'status': f'API Error: {response.status_code}'
                            })

                        total_checked += 1

                        # Rate limiting - be respectful to API
                        if total_checked % 10 == 0:
                            print(f"Checked {total_checked} addresses...")
                            time.sleep(1)

                    except Exception as e:
                        writer.writerow({
                            'public_key': row.get('Public Key', ''),
                            'private_key': row.get('Private Key (WIF)', ''),
                            'address': address,
                            'balance_btc': 0,
                            'balance_usd': 0,
                            'status': f'Error: {str(e)}'
                        })
                        continue

        print(f"\n📊 BALANCE CHECK COMPLETE")
        print(f"✅ Total addresses checked: {total_checked}")
        print(f"💰 Addresses with balance: {addresses_with_balance}")

        if balances_found:
            print(f"\n🎉 FOUND {len(balances_found)} ADDRESSES WITH FUNDS:")
            total_btc = sum(item['balance_btc'] for item in balances_found)
            total_usd = sum(item['balance_usd'] for item in balances_found if 'balance_usd' in item)
            print(f"Total BTC found: {total_btc:.8f}")
            if total_usd > 0:
                print(f"Approximate USD value: ${total_usd:.2f}")

            # Save detailed report
            detailed_report = f"balances_detailed_{datetime.now().strftime('%Y%m%d_%H%M%S')}.csv"
            with open(detailed_report, 'w', newline='') as f:
                fieldnames = ['address', 'balance_btc', 'balance_usd', 'timestamp']
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writeheader()
                for item in balances_found:
                    item['timestamp'] = datetime.now().isoformat()
                    writer.writerow(item)

            print(f"💾 Detailed report saved to: {detailed_report}")

        return balances_found

    except Exception as e:
        print(f"❌ Error checking balances: {e}")
        return []

import hashlib
import base58
import csv

def convert_uncompressed_to_compressed(uncompressed_wif):
    """Convert uncompressed WIF to compressed WIF"""
    try:
        # Decode uncompressed WIF
        decoded = base58.b58decode(uncompressed_wif)
        priv_key_bytes = decoded[1:-4]  # Remove version and checksum

        # Add compression flag (0x01)
        compressed_priv_key = priv_key_bytes + b'\x01'

        # Add version byte (0x80 for mainnet)
        extended_key = b'\x80' + compressed_priv_key

        # Calculate checksum
        checksum = hashlib.sha256(hashlib.sha256(extended_key).digest()).digest()[:4]

        # Encode
        compressed_wif_bytes = extended_key + checksum
        return base58.b58encode(compressed_wif_bytes).decode('utf-8')

    except:
        return None

import hashlib
import base58
import ecdsa
import json
import sys
import os
from datetime import datetime
import binascii

class KeystoreAnalyzer:
    def __init__(self, file_path):
        self.file_path = file_path
        self.output_dir = f"keystore_export_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        os.makedirs(self.output_dir, exist_ok=True)

    def load_keystore(self):
        """Load the keystore with error handling"""
        try:
            with open(self.file_path, 'r') as file:
                content = file.read()

                # Try to find valid JSON at the beginning
                try:
                    data = json.loads(content)
                    return data
                except json.JSONDecodeError as e:
                    print(f"JSON decode error: {e}")
                    print("Trying to extract valid JSON portion...")

                    # Try to find the valid portion
                    lines = content.split('\n')
                    valid_lines = []

                    for i, line in enumerate(lines):
                        valid_lines.append(line)
                        try:
                            test_content = '\n'.join(valid_lines)
                            if test_content.strip():
                                json.loads(test_content)
                        except json.JSONDecodeError:
                            if i > 0:  # Remove the last problematic line
                                valid_lines.pop()
                                break

                    if valid_lines:
                        valid_content = '\n'.join(valid_lines)
                        return json.loads(valid_content)
                    else:
                        raise Exception("No valid JSON found")

        except Exception as e:
            print(f"Error loading keystore: {e}")

            # Try manual parsing for your specific format
            print("Attempting manual parsing...")
            return self.manual_parse_keystore()

        return None

    def manual_parse_keystore(self):
        """Manually parse keystore when JSON fails"""
        try:
            with open(self.file_path, 'r') as file:
                content = file.read()

            # Look for keypairs manually
            keypairs = {}

            # Find keystore section
            if '"keystore"' in content:
                # Try to extract keypairs section
                start_marker = '"keypairs": {'
                end_marker = '},'

                start_pos = content.find(start_marker)
                if start_pos != -1:
                    # Find the end of keypairs section
                    end_pos = content.find(end_marker, start_pos)
                    if end_pos == -1:
                        end_pos = content.find('}}', start_pos)  # Alternative ending

                    if end_pos != -1:
                        keypairs_section = content[start_pos + len(start_marker):end_pos]

                        # Parse keypairs manually
                        # This is a simplified parser - adjust based on your exact format
                        entries = keypairs_section.split('",')

                        for entry in entries:
                            if '":"' in entry:
                                parts = entry.split('":"')
                                if len(parts) >= 2:
                                    pubkey = parts[0].strip().strip('"')
                                    privkey = parts[1].strip().strip('"')
                                    if pubkey and privkey and len(pubkey) > 50:
                                        keypairs[pubkey] = privkey

            return {"keystore": {"keypairs": keypairs}}

        except Exception as e:
            print(f"Manual parsing failed: {e}")
            return None

    def wif_to_hex(self, wif):
        """Convert WIF to hex"""
        try:
            decoded = base58.b58decode(wif)
            return decoded[1:-4].hex()  # Remove version and checksum
        except:
            return None

    def derive_address(self, pub_key):
        """Derive Bitcoin address from public key"""
        try:
            pub_key_clean = pub_key.replace('0x', '').strip()

            # Handle format
            if pub_key_clean.startswith('04') and len(pub_key_clean) >= 130:
                pub_key_clean = pub_key_clean[:130]  # Truncate if too long
            elif len(pub_key_clean) >= 128:
                pub_key_clean = '04' + pub_key_clean[:128]  # Take first 128 chars

            # SHA256
            sha256_hash = hashlib.sha256(bytes.fromhex(pub_key_clean)).digest()

            # RIPEMD160
            ripemd160_hash = hashlib.new('ripemd160', sha256_hash).digest()

            # Version byte + checksum
            extended = b'\x00' + ripemd160_hash
            checksum = hashlib.sha256(hashlib.sha256(extended).digest()).digest()[:4]

            # Encode
            address_bytes = extended + checksum
            return base58.b58encode(address_bytes).decode('utf-8')
        except Exception as e:
            print(f"Address derivation error for key {pub_key[:20]}...: {e}")
            return None

    def export_all_keys(self):
        """Export all keypairs in various formats"""
        data = self.load_keystore()
        if not data:
            print("Failed to load keystore data")
            return False

        print("Successfully loaded keystore data")

        keypairs = []

        # Extract from keystore.keypairs
        keystore_data = data.get('keystore', {})
        if isinstance(keystore_data, dict) and 'keypairs' in keystore_data:
            keypairs_dict = keystore_data['keypairs']
            print(f"Found {len(keypairs_dict)} keypairs in keystore")

            for pubkey, privkey in keypairs_dict.items():
                if pubkey and privkey:  # Skip empty entries
                    try:
                        derived_addr = self.derive_address(pubkey)
                        keypairs.append({
                            'public_key': pubkey,
                            'private_key_wif': privkey,
                            'private_key_hex': self.wif_to_hex(privkey),
                            'derived_address': derived_addr
                        })
                    except Exception as e:
                        print(f"Error processing keypair: {e}")
                        continue
        else:
            print("No keypairs found in standard location, checking alternative locations...")

            # Check other possible locations
            for key, value in data.items():
                if isinstance(value, dict) and 'keypairs' in value:
                    print(f"Found keypairs in '{key}' section")
                    for pubkey, privkey in value['keypairs'].items():
                        if pubkey and privkey:
                            try:
                                derived_addr = self.derive_address(pubkey)
                                keypairs.append({
                                    'public_key': pubkey,
                                    'private_key_wif': privkey,
                                    'private_key_hex': self.wif_to_hex(privkey),
                                    'derived_address': derived_addr
                                })
                            except Exception as e:
                                print(f"Error processing keypair: {e}")
                                continue

        if not keypairs:
            print("No valid keypairs found!")
            return False

        # Export to various formats
        self.export_csv(keypairs)
        self.export_json(keypairs)
        self.export_txt_summary(keypairs)

        print(f"\n✅ Export completed!")
        print(f"📁 Files saved to: {self.output_dir}")
        print(f"📊 Total keypairs exported: {len(keypairs)}")

        return True

    def export_csv(self, keypairs):
        """Export keypairs to CSV"""
        csv_file = os.path.join(self.output_dir, "keypairs.csv")
        with open(csv_file, 'w') as f:
            f.write("Public Key,Private Key (WIF),Private Key (HEX),Derived Address\n")
            for kp in keypairs:
                f.write(f"{kp['public_key']},{kp['private_key_wif']},{kp['private_key_hex'] or ''},{kp['derived_address'] or ''}\n")

    def export_json(self, keypairs):
        """Export keypairs to JSON"""
        json_file = os.path.join(self.output_dir, "keypairs.json")
        with open(json_file, 'w') as f:
            json.dump(keypairs, f, indent=2)

    def export_txt_summary(self, keypairs):
        """Export summary to TXT"""
        txt_file = os.path.join(self.output_dir, "keypairs_summary.txt")
        with open(txt_file, 'w') as f:
            f.write("KEYPAIRS SUMMARY\n")
            f.write("="*50 + "\n")
            f.write(f"Generated on: {datetime.now()}\n")
            f.write(f"Total keypairs: {len(keypairs)}\n\n")

            for i, kp in enumerate(keypairs, 1):
                f.write(f"--- Key Pair #{i} ---\n")
                f.write(f"Public Key:  {kp['public_key']}\n")
                f.write(f"Private Key: {kp['private_key_wif']}\n")
                f.write(f"Address:     {kp['derived_address']}\n\n")

# format_segwit_keys.py
import hashlib
import base58
import bech32
import ecdsa
import sys
import os

def sha256(data):
    """SHA256 hash"""
    return hashlib.sha256(data).digest()

def hash160(data):
    """RIPEMD160(SHA256(data))"""
    return hashlib.new('ripemd160', sha256(data)).digest()

def wif_to_private_key(wif):
    """Convert WIF to private key bytes"""
    try:
        decoded = base58.b58decode(wif)
        priv_key_bytes = decoded[1:-4]

        # Check if compressed
        if len(priv_key_bytes) == 33 and priv_key_bytes[-1] == 1:
            priv_key_bytes = priv_key_bytes[:-1]

        return priv_key_bytes
    except:
        return None

def private_key_to_public_key(private_key_bytes):
    """Derive compressed public key from private key"""
    try:
        sk = ecdsa.SigningKey.from_string(private_key_bytes, curve=ecdsa.SECP256k1)
        vk = sk.get_verifying_key()

        # Compressed public key
        x_coord = vk.to_string()[:32]
        y_parity = vk.to_string()[32] & 1
        prefix = b'\x03' if y_parity else b'\x02'
        return prefix + x_coord
    except:
        return None

def create_p2wpkh_address(public_key_bytes):
    """Create P2WPKH address (bech32)"""
    try:
        pubkey_hash = hash160(public_key_bytes)
        pubkey_hash_5bit = bech32.convertbits(pubkey_hash, 8, 5)
        address_words = [0] + pubkey_hash_5bit
        return bech32.bech32_encode('bc', address_words)
    except:
        return "conversion_failed"

def create_p2wpkh_p2sh_address(public_key_bytes):
    """Create P2WPKH-P2SH address"""
    try:
        pubkey_hash = hash160(public_key_bytes)
        witness_program = b'\x00\x14' + pubkey_hash
        script_hash = hash160(witness_program)
        versioned_hash = b'\x05' + script_hash
        checksum = sha256(sha256(versioned_hash))[:4]
        address_bytes = versioned_hash + checksum
        return base58.b58encode(address_bytes).decode('utf-8')
    except:
        return "conversion_failed"

def process_keys_file(input_file, output_file):
    """Process keys file and create formatted output"""

    print(f"🔄 Processing keys from: {input_file}")

    # Read keys
    keys = []
    if input_file.endswith('.csv'):
        import csv
        with open(input_file, 'r') as f:
            reader = csv.DictReader(f)
            for row in reader:
                privkey = row.get('Private Key (WIF)', '').strip()
                if privkey.startswith(('K', 'L')) and len(privkey) > 30:
                    keys.append(privkey)
    else:
        with open(input_file, 'r') as f:
            for line in f:
                privkey = line.strip()
                if privkey.startswith(('K', 'L')) and len(privkey) > 30:
                    keys.append(privkey)

    print(f"✅ Found {len(keys)} K/L keys to process")

    # Process each key
    formatted_lines = []
    processed_count = 0

    for i, privkey in enumerate(keys):
        try:
            # Convert WIF to private key bytes
            priv_key_bytes = wif_to_private_key(privkey)
            if not priv_key_bytes:
                continue

            # Derive public key
            pub_key_bytes = private_key_to_public_key(priv_key_bytes)
            if not pub_key_bytes:
                continue

            # Generate both address types
            p2wpkh_addr = create_p2wpkh_address(pub_key_bytes)
            p2wpkh_p2sh_addr = create_p2wpkh_p2sh_address(pub_key_bytes)

            # Format as requested
            formatted_lines.append(f"p2wpkh:{privkey}")
            formatted_lines.append(f"p2wpkh-p2sh:{privkey}")

            processed_count += 1

            # Show progress
            if processed_count <= 5:
                print(f"   Processed: {privkey[:20]}...")
            elif processed_count == 6:
                print("   ... (continuing)")

        except Exception as e:
            print(f"   Error processing key: {e}")
            continue

        # Limit for testing - remove this for full processing
        if processed_count >= 100:
            print("   Stopping early for demo...")
            break

    # Save formatted output
    with open(output_file, 'w') as f:
        for line in formatted_lines:
            f.write(f"{line}\n")

    print(f"\n✅ Processing Complete!")
    print(f"   Total keys processed: {processed_count}")
    print(f"   Output saved to: {output_file}")
    print(f"\n📋 Sample output:")
    for line in formatted_lines[:6]:  # Show first 3 pairs
        print(f"   {line}")

    return formatted_lines

import re
import json
import os
from datetime import datetime

def extract_keypairs_improved(filename):
    """Improved keypair extraction with better pattern matching"""
    print(f"🔍 Extracting keypairs from: {filename}")

    try:
        # Read in chunks to handle large files
        print("Reading large file...")
        with open(filename, 'rb') as f:
            content = f.read()

        # Convert to string (handle encoding issues)
        try:
            text = content.decode('utf-8')
        except UnicodeDecodeError:
            text = content.decode('utf-8', errors='ignore')

        print(f"File size: {len(text)} characters")

        # Multiple pattern approaches for Electrum-style keypairs

        # Pattern 1: Standard Electrum format
        print("Searching for standard Electrum format...")
        pattern1 = r'"([0-9a-fA-F]{128,130})"\s*:\s*"([5KL][1-9A-HJ-NP-Za-km-z]{40,60})"'
        matches1 = re.findall(pattern1, text)

        # Pattern 2: With 0x prefix or 04 prefix
        print("Searching for prefixed formats...")
        pattern2 = r'"(0[4x][0-9a-fA-F]{126,128})"\s*:\s*"([5KL][1-9A-HJ-NP-Za-km-z]{40,60})"'
        matches2 = re.findall(pattern2, text)

        # Pattern 3: More flexible hex pattern
        print("Searching with flexible patterns...")
        pattern3 = r'"([0-9a-fA-F]{100,150})"\s*:\s*"([5KL][1-9A-HJ-NP-Za-km-z]{40,60})"'
        matches3 = re.findall(pattern3, text)

        # Combine all matches
        all_matches = matches1 + matches2 + matches3

        # Remove duplicates and validate

def derive_simple_address(pubkey):
    """Simple address derivation (basic implementation)"""
    try:
        import hashlib
        import base58

        # Clean pubkey
        pubkey_clean = pubkey.replace('0x', '').strip()
        if len(pubkey_clean) == 128:
            pubkey_clean = '04' + pubkey_clean
        elif not pubkey_clean.startswith('04') or len(pubkey_clean) != 130:
            return None

        # SHA256
        sha256_hash = hashlib.sha256(bytes.fromhex(pubkey_clean)).digest()

        # RIPEMD160
        ripemd160 = hashlib.new('ripemd160')
        ripemd160.update(sha256_hash)
        ripemd160_hash = ripemd160.digest()

        # Version byte + checksum
        versioned = b'\x00' + ripemd160_hash
        checksum = hashlib.sha256(hashlib.sha256(versioned).digest()).digest()[:4]

        # Encode
        address_bytes = versioned + checksum
        return base58.b58encode(address_bytes).decode('utf-8')
    except:
        return "Derivation failed"

def sha256(data):
    """SHA256 hash"""
    return hashlib.sha256(data).digest()

def hash160(data):
    """RIPEMD160(SHA256(data))"""
    return hashlib.new('ripemd160', sha256(data)).digest()

def wif_to_private_key(wif):
    """Convert WIF to private key bytes"""
    try:
        decoded = base58.b58decode(wif)
        # Remove version byte (first) and checksum (last 4 bytes)
        priv_key_bytes = decoded[1:-4]

        # Check if compressed (has 0x01 suffix)
        if len(priv_key_bytes) == 33 and priv_key_bytes[-1] == 1:
            priv_key_bytes = priv_key_bytes[:-1]  # Remove compression flag

        return priv_key_bytes
    except:
        return None

def private_key_to_public_key(private_key_bytes, compressed=True):
    """Derive public key from private key"""
    try:
        sk = ecdsa.SigningKey.from_string(private_key_bytes, curve=ecdsa.SECP256k1)
        vk = sk.get_verifying_key()

        if compressed:
            # Compressed public key: 02|03 + x-coordinate
            x_coord = vk.to_string()[:32]
            # Check y-coordinate parity for prefix
            y_parity = vk.to_string()[32] & 1
            prefix = b'\x03' if y_parity else b'\x02'
            return prefix + x_coord
        else:
            # Uncompressed public key: 04 + x + y
            return b'\x04' + vk.to_string()
    except:
        return None

def create_p2wpkh_address(public_key_bytes):
    """Create P2WPKH address (bech32, starts with bc1)"""
    try:
        # P2WPKH uses HASH160 of compressed public key
        pubkey_hash = hash160(public_key_bytes)

        # For simplicity, returning placeholder - real implementation needs bech32 encoding
        # In practice, you'd use a bech32 library
        return f"bc1_address_for_{pubkey_hash.hex()[:16]}"
    except:
        return "conversion_failed"

def create_p2wpkh_p2sh_address(public_key_bytes):
    """Create P2WPKH-P2SH address (starts with 3)"""
    try:
        # P2WPKH script: OP_0 + PUSH20 + pubkey_hash
        pubkey_hash = hash160(public_key_bytes)
        witness_program = b'\x00\x14' + pubkey_hash  # OP_0 (version 0) + PUSH20 + 20-byte hash

        # P2SH wraps the witness program
        script_hash = hash160(witness_program)

        # P2SH version byte is 0x05
        versioned_hash = b'\x05' + script_hash
        checksum = sha256(sha256(versioned_hash))[:4]
        address_bytes = versioned_hash + checksum

        return base58.b58encode(address_bytes).decode('utf-8')
    except:
        return "conversion_failed"

        # Remove version byte (first) and checksum (last 4 bytes)
        priv_key_bytes = decoded[1:-4]

        # Check if compressed (has 0x01 suffix)
        if len(priv_key_bytes) == 33 and priv_key_bytes[-1] == 1:
            priv_key_bytes = priv_key_bytes[:-1]  # Remove compression flag

        return priv_key_bytes
    except Exception as e:
        print(f"WIF conversion error: {e}")
        return None

def private_key_to_public_key(private_key_bytes, compressed=True):
    """Derive public key from private key"""
    try:
        sk = ecdsa.SigningKey.from_string(private_key_bytes, curve=ecdsa.SECP256k1)
        vk = sk.get_verifying_key()

        if compressed:
            # Compressed public key: 02|03 + x-coordinate
            x_coord = vk.to_string()[:32]
            # Check y-coordinate parity for prefix
            y_parity = vk.to_string()[32] & 1
            prefix = b'\x03' if y_parity else b'\x02'
            return prefix + x_coord
        else:
            # Uncompressed public key: 04 + x + y
            return b'\x04' + vk.to_string()
    except Exception as e:
        print(f"Public key derivation error: {e}")
        return None

def create_p2wpkh_address(public_key_bytes, network='mainnet'):
    """Create P2WPKH address (bech32, starts with bc1)"""
    try:
        # P2WPKH uses HASH160 of compressed public key
        pubkey_hash = hash160(public_key_bytes)

        # Convert to 5-bit format for bech32
        pubkey_hash_5bit = bech32.convertbits(pubkey_hash, 8, 5)

        # Create bech32 address
        hrp = 'bc' if network == 'mainnet' else 'tb'
        address_words = [0] + pubkey_hash_5bit  # 0 = witness version
        address = bech32.bech32_encode(hrp, address_words)

        return address
    except Exception as e:
        print(f"P2WPKH creation error: {e}")
        return "conversion_failed"

def create_p2wpkh_p2sh_address(public_key_bytes):
    """Create P2WPKH-P2SH address (starts with 3)"""
    try:
        # P2WPKH script: OP_0 + PUSH20 + pubkey_hash
        pubkey_hash = hash160(public_key_bytes)
        witness_program = b'\x00\x14' + pubkey_hash  # OP_0 (version 0) + PUSH20 + 20-byte hash

        # P2SH wraps the witness program
        script_hash = hash160(witness_program)

        # P2SH version byte is 0x05
        versioned_hash = b'\x05' + script_hash
        checksum = sha256(sha256(versioned_hash))[:4]
        address_bytes = versioned_hash + checksum

        return base58.b58encode(address_bytes).decode('utf-8')
    except Exception as e:
        print(f"P2WPKH-P2SH creation error: {e}")
        return "conversion_failed"

def generate_segwit_addresses(input_file):

            writer.writeheader()
            writer.writerows(results)

        # Save addresses only

                f.write(f"{result['p2wpkh-p2sh_address']}\n")

        print(f"💾 Results saved to:")
        print(f"   CSV file: {csv_output}")
        print(f"   Addresses file: {txt_output}")

            print(f"   {i}. P2WPKH-P2SH: {result['p2wpkh-p2sh_address']}")

    return results

import hashlib
import base58
import ecdsa
import json
import sys
import os
from datetime import datetime

class KeyPairingBot:
    def __init__(self, file_path):
        self.file_path = file_path
        self.output_file = f"keypairing_results_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"

    def load_keys_from_file(self):
        """Load keypairs from Electrum wallet format"""
        keys_data = []
        try:
            with open(self.file_path, 'r') as file:
                data = json.load(file)

                # Look for keystore -> keypairs section (Electrum format)
                if 'keystore' in data and 'keypairs' in data['keystore']:
                    keypairs = data['keystore']['keypairs']
                    for pubkey, privkey in keypairs.items():
                        key_pair = {
                            'pubkey': pubkey,
                            'privkey': privkey,
                            'source': 'keystore_keypairs'
                        }
                        keys_data.append(key_pair)

                # Also look for addresses section with additional data
                if 'addresses' in data:
                    for address, addr_data in data['addresses'].items():
                        if isinstance(addr_data, dict) and 'pubkey' in addr_data:
                            key_pair = {
                                'address': address,
                                'pubkey': addr_data['pubkey'],
                                'type': addr_data.get('type', 'unknown'),
                                'source': 'addresses_section'
                            }
                            keys_data.append(key_pair)

        except Exception as e:
            print(f"Error loading file: {e}")
            return []

        return keys_data

    def clean_key(self, key):
        """Clean key string by removing common prefixes and formatting"""
        if not key:
            return ""
        key = str(key).strip().strip('"\'').replace("0x", "").strip()
        return key

    def validate_key_pair(self, priv_key, pub_key):
        """Validate if private key corresponds to public key"""
        try:
            priv_key_clean = self.clean_key(priv_key)
            pub_key_clean = self.clean_key(pub_key)

            if not priv_key_clean or not pub_key_clean:
                return False

            # Convert WIF private key to hex if needed
            if priv_key_clean.startswith('5') or priv_key_clean.startswith('K') or priv_key_clean.startswith('L'):
                # WIF format - convert to hex
                priv_key_hex = self.wif_to_hex(priv_key_clean)
                if not priv_key_hex:
                    return False
                priv_key_bytes = bytes.fromhex(priv_key_hex)
            else:
                # Already hex format
                priv_key_bytes = bytes.fromhex(priv_key_clean)

            # Create signing key from private key
            sk = ecdsa.SigningKey.from_string(priv_key_bytes, curve=ecdsa.SECP256k1)
            vk = sk.get_verifying_key()

            # Get public key in uncompressed format
            pub_key_computed = '04' + vk.to_string().hex()

            # Compare with provided public key
            return pub_key_computed.lower() == pub_key_clean.lower()
        except Exception as e:
            print(f"Validation error: {e}")
            return False

    def wif_to_hex(self, wif):
        """Convert WIF private key to hex"""
        try:
            # Decode from base58
            decoded = base58.b58decode(wif)
            # Remove version byte (first) and checksum (last 4 bytes)
            priv_key_bytes = decoded[1:-4]
            return priv_key_bytes.hex()
        except:
            return None

    def derive_legacy_address(self, pub_key):
        """Derive legacy Bitcoin address from public key"""
        try:
            pub_key_clean = self.clean_key(pub_key)

            if not pub_key_clean:
                return None

            # Ensure proper format
            if pub_key_clean.startswith('04') and len(pub_key_clean) == 130:
                pass  # Good format
            elif len(pub_key_clean) == 128:
                pub_key_clean = '04' + pub_key_clean
            else:
                return None

            # SHA256 hash of public key
            sha256_hash = hashlib.sha256(bytes.fromhex(pub_key_clean)).digest()

            # RIPEMD160 hash
            ripemd160_hash = hashlib.new('ripemd160', sha256_hash).digest()

            # Add version byte (0x00 for mainnet)
            extended_ripemd160 = b'\x00' + ripemd160_hash

            # Double SHA256 for checksum
            checksum = hashlib.sha256(hashlib.sha256(extended_ripemd160).digest()).digest()[:4]

            # Combine and encode in base58
            address_bytes = extended_ripemd160 + checksum
            legacy_address = base58.b58encode(address_bytes).decode('utf-8')

            return legacy_address
        except Exception as e:
            print(f"Address derivation error: {e}")
            return None

    def pair_keys(self):
        """Main function to pair all keys"""
        key_data = self.load_keys_from_file()
        if not key_data:
            print("No key data found in file")
            return []

        results = []

        for i, data in enumerate(key_data):
            pubkey = data.get('pubkey', 'N/A')
            privkey = data.get('privkey', 'N/A') or 'N/A'
            address = data.get('address', 'N/A')
            source = data.get('source', 'unknown')

            result = {
                'pair_id': i + 1,
                'pubkey': pubkey,
                'privkey': privkey,
                'legacy_address': address,
                'source': source,
                'validation': {}
            }

            # Validate key pairing if both keys exist
            if privkey != 'N/A' and pubkey != 'N/A' and privkey and pubkey:
                is_valid = self.validate_key_pair(privkey, pubkey)
                result['validation']['keys_match'] = is_valid

            # Derive address from public key
            if pubkey != 'N/A' and pubkey:
                derived_address = self.derive_legacy_address(pubkey)
                if derived_address:
                    result['validation']['derived_address'] = derived_address
                    # If original address exists, compare
                    if address != 'N/A' and address:
                        matches = derived_address == address
                        result['validation']['address_match'] = matches

            results.append(result)

        return results

    def save_results(self, results):
        """Save results to a new file"""
        try:
            with open(self.output_file, 'w') as f:
                f.write(f"KEY PAIRING RESULTS FOR: {self.file_path}\n")
                f.write(f"Generated on: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
                f.write("="*80 + "\n\n")

                if not results:
                    f.write("No valid key pairs found!\n")
                    return True

                valid_pairs = 0
                invalid_pairs = 0

                for result in results:
                    f.write(f"--- Pair #{result['pair_id']} (Source: {result['source']}) ---\n")
                    f.write(f"Public Key:  {str(result['pubkey'])[:100]}...\n" if len(str(result['pubkey'])) > 100 else f"Public Key:  {result['pubkey']}\n")
                    f.write(f"Private Key: {str(result['privkey'])[:100]}...\n" if len(str(result['privkey'])) > 100 else f"Private Key: {result['privkey']}\n")
                    if result['legacy_address'] != 'N/A':
                        f.write(f"Legacy Addr: {result['legacy_address']}\n")

                    if result['validation']:
                        f.write("Validation:\n")
                        for key, value in result['validation'].items():
                            if key == 'derived_address':
                                f.write(f"  Derived Address: {value}\n")
                            else:
                                status = "✓ PASS" if value else "✗ FAIL"
                                f.write(f"  {key.replace('_', ' ').title()}: {status}\n")
                                if key == 'keys_match':
                                    if value:
                                        valid_pairs += 1
                                    else:
                                        invalid_pairs += 1
                    else:
                        f.write("  No validation data available\n")
                    f.write("\n")

                f.write(f"Summary:\n")
                f.write(f"  Valid pairs: {valid_pairs}\n")
                f.write(f"  Invalid pairs: {invalid_pairs}\n")
                f.write(f"  Total pairs processed: {len(results)}\n")

            print(f"Results saved to: {self.output_file}")
            return True
        except Exception as e:
            print(f"Error saving results: {e}")
            return False

    def print_results(self, results):
        """Print formatted results to console"""
        print(f"\n{'='*80}")
        print(f"KEY PAIRING RESULTS FOR: {self.file_path}")
        print(f"{'='*80}")

        if not results:
            print("No valid key pairs found!")
            return

        valid_pairs = 0
        invalid_pairs = 0

        for result in results:
            print(f"\n--- Pair #{result['pair_id']} (Source: {result['source']}) ---")
            pubkey_str = str(result['pubkey'])
            privkey_str = str(result['privkey'])
            print(f"Public Key:  {pubkey_str[:80]}..." if len(pubkey_str) > 80 else f"Public Key:  {pubkey_str}")
            print(f"Private Key: {privkey_str[:80]}..." if len(privkey_str) > 80 else f"Private Key: {privkey_str}")
            if result['legacy_address'] != 'N/A':
                print(f"Legacy Addr: {result['legacy_address']}")

            if result['validation']:
                print("Validation:")
                for key, value in result['validation'].items():
                    if key == 'derived_address':
                        print(f"  Derived Address: {value}")
                    else:
                        status = "✓ PASS" if value else "✗ FAIL"
                        print(f"  {key.replace('_', ' ').title()}: {status}")
                        if key == 'keys_match':
                            if value:
                                valid_pairs += 1
                            else:
                                invalid_pairs += 1
            else:
                print("  No validation data available")

def main():
    # Check if file argument is provided
    if len(sys.argv) != 2:
        print("Usage: python keypairingbot.py <filename>")

import base58
import hashlib

def convert_uncompressed_to_compressed_wif(uncompressed_wif):
    """Convert uncompressed WIF to compressed WIF"""
    try:
        # Decode the uncompressed WIF
        decoded = base58.b58decode(uncompressed_wif)

        # Extract private key (remove version byte and checksum)
        priv_key_bytes = decoded[1:-4]

        # Add compression flag (0x01)
        compressed_priv_key = priv_key_bytes + b'\x01'

        # Add version byte (0x80 for mainnet)
        extended_key = b'\x80' + compressed_priv_key

        # Calculate checksum
        checksum = hashlib.sha256(hashlib.sha256(extended_key).digest()).digest()[:4]

        # Create final compressed WIF
        compressed_wif_bytes = extended_key + checksum
        compressed_wif = base58.b58encode(compressed_wif_bytes).decode('utf-8')

        return compressed_wif

    except Exception as e:
        print(f"Conversion error: {e}")
        return None

def batch_convert_keys(input_file, output_file):
    """Convert all uncompressed keys to compressed format"""
    print(f"🔄 Converting keys from {input_file}...")

    converted_count = 0
    converted_keys = []

    try:
        with open(input_file, 'r') as f:
            # Read first few lines to determine format
            first_line = f.readline()
            f.seek(0)

            if 'Private Key' in first_line:
                # CSV format
                import csv
                reader = csv.DictReader(f)
                for row in reader:
                    uncompressed_wif = row.get('Private Key (WIF)', '').strip()
                    if uncompressed_wif.startswith('5'):
                        compressed_wif = convert_uncompressed_to_compressed_wif(uncompressed_wif)
                        if compressed_wif:
                            converted_keys.append({
                                'original': uncompressed_wif,
                                'compressed': compressed_wif
                            })
                            converted_count += 1

                            # Show progress
                            if converted_count <= 5:
                                print(f"   {uncompressed_wif[:15]}... → {compressed_wif[:15]}...")
            else:
                # Simple text format (one key per line)
                for line in f:
                    uncompressed_wif = line.strip()
                    if uncompressed_wif.startswith('5'):
                        compressed_wif = convert_uncompressed_to_compressed_wif(uncompressed_wif)
                        if compressed_wif:
                            converted_keys.append({
                                'original': uncompressed_wif,
                                'compressed': compressed_wif
                            })
                            converted_count += 1

        # Save results
        if converted_keys:
            with open(output_file, 'w') as f:
                f.write("original_uncompressed,compressed_wif\n")
                for key_pair in converted_keys:
                    f.write(f"{key_pair['original']},{key_pair['compressed']}\n")

            # Also save just compressed keys
            compressed_only_file = output_file.replace('.csv', '_compressed_only.txt')
            with open(compressed_only_file, 'w') as f:
                for key_pair in converted_keys:
                    f.write(f"{key_pair['compressed']}\n")

            print(f"\n✅ Conversion Complete!")
            print(f"   Total converted: {converted_count}")
            print(f"   Full mapping saved to: {output_file}")
            print(f"   Compressed keys only: {compressed_only_file}")
        else:
            print("❌ No keys were converted")

    except Exception as e:
        print(f"❌ Error in batch conversion: {e}")

import json

def repair_corrupted_keystore(input_file, output_file=None):
    """Attempt to repair corrupted keystore file"""

    if output_file is None:
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        output_file = f"repaired_wallet_{timestamp}.json"

    print(f"🔧 Attempting to repair: {input_file}")
    print(f"💾 Will save to: {output_file}")

    try:
        # Read the entire file
        with open(input_file, 'rb') as f:
            content = f.read()

        # Convert to string, handling encoding issues
        try:
            text_content = content.decode('utf-8')
        except UnicodeDecodeError:
            text_content = content.decode('utf-8', errors='ignore')

        print(f"📄 File size: {len(text_content)} characters")

        # Method 1: Try to find valid JSON structure
        repaired_content = None

        # Look for the keystore section
        keystore_start = text_content.find('"keystore"')
        if keystore_start == -1:
            keystore_start = text_content.find('keystore')

        if keystore_start != -1:
            print("✅ Found keystore section")

            # Look for keypairs section within keystore
            keypairs_start = text_content.find('"keypairs"', keystore_start)
            if keypairs_start != -1:
                print("✅ Found keypairs section")

                # Find the start of keypairs data
                keypairs_actual_start = text_content.find('{', keypairs_start)
                if keypairs_actual_start != -1:
                    # Try to balance braces to find end
                    brace_count = 1
                    pos = keypairs_actual_start + 1

                    while pos < len(text_content) and brace_count > 0:
                        if text_content[pos] == '{':
                            brace_count += 1
                        elif text_content[pos] == '}':
                            brace_count -= 1
                        pos += 1

                    if brace_count == 0:
                        keypairs_end = pos
                        print(f"✅ Found balanced braces ending at position {keypairs_end}")

                        # Extract the keypairs section
                        before_keypairs = text_content[:keypairs_actual_start + 1]
                        keypairs_content = text_content[keypairs_actual_start:keypairs_end]
                        after_keypairs = text_content[keypairs_end:]

                        # Try to parse just the keypairs content
                        try:
                            # Clean up the keypairs content
                            cleaned_keypairs = clean_keypairs_section(keypairs_content)

                            # Reconstruct the full JSON
                            reconstructed = before_keypairs + cleaned_keypairs + after_keypairs

                            # Try to parse the reconstructed content
                            json.loads(reconstructed)
                            repaired_content = reconstructed
                            print("✅ Successfully reconstructed JSON")

                        except json.JSONDecodeError as e:
                            print(f"❌ Reconstruction failed: {e}")
                            # Try simpler approach
                            repaired_content = attempt_simple_repair(text_content)
                    else:
                        print("❌ Could not balance braces, trying simple repair")
                        repaired_content = attempt_simple_repair(text_content)
                else:
                    print("❌ Could not find keypairs start, trying simple repair")
                    repaired_content = attempt_simple_repair(text_content)
            else:
                print("❌ No keypairs section found, trying simple repair")
                repaired_content = attempt_simple_repair(text_content)
        else:
            print("❌ No keystore section found, trying simple repair")
            repaired_content = attempt_simple_repair(text_content)

        # If we couldn't repair using smart method, try brute force
        if repaired_content is None:
            print("🔄 Trying brute force repair...")
            repaired_content = brute_force_repair(text_content)

        # Save the repaired content
        if repaired_content:
            with open(output_file, 'w', encoding='utf-8') as f:
                f.write(repaired_content)

            # Test if it's valid JSON now
            try:
                test_data = json.loads(repaired_content)
                print("✅ Repair successful! File is now valid JSON")
                print(f"📦 Keystore contains {count_keypairs(test_data)} keypairs")
                return output_file
            except json.JSONDecodeError as e:
                print(f"❌ Repaired file still not valid JSON: {e}")
                # Save anyway for manual inspection
                with open(output_file + ".attempt", 'w', encoding='utf-8') as f:
                    f.write(repaired_content)
                print(f"📝 Saved repair attempt to {output_file}.attempt")
                return None
        else:
            print("❌ Could not repair the file")
            return None

    except Exception as e:
        print(f"❌ Error during repair: {e}")
        return None

def clean_keypairs_section(content):
    """Clean up keypairs section"""
    # Remove trailing commas before closing braces
    content = re.sub(r',(\s*[}\]])', r'\1', content)

    # Fix common JSON issues
    content = content.replace('\\"', '"')  # Fix escaped quotes

    # Ensure proper JSON formatting
    lines = content.split('\n')
    cleaned_lines = []

    for line in lines:
        line = line.strip()
        if line and not line.startswith('//'):  # Remove comments
            cleaned_lines.append(line)

    return ''.join(cleaned_lines)

def attempt_simple_repair(content):
    """Simple repair by truncating at first error"""
    try:
        # Try to parse incrementally
        for i in range(len(content), 0, -100):
            test_content = content[:i]
            try:
                json.loads(test_content)
                print(f"✅ Valid JSON found up to position {i}")
                return test_content
            except:
                continue
        return None
    except:
        return None

def brute_force_repair(content):
    """Brute force approach to extract keypairs"""
    print("🔨 Using brute force extraction...")

    # Try to extract keypairs manually
    keypairs = {}

    # Look for patterns like "pubkey": "privkey"
    pattern = r'"([^"]{100,})"\s*:\s*"([^"]+)"'
    matches = re.findall(pattern, content)

    for pubkey, privkey in matches:
        if len(pubkey) > 100 and len(privkey) > 20:  # Likely valid keys
            keypairs[pubkey] = privkey

    if keypairs:
        print(f"✅ Extracted {len(keypairs)} keypairs via brute force")
        # Create minimal valid JSON
        minimal_json = {
            "keystore": {
                "keypairs": keypairs
            }
        }
        return json.dumps(minimal_json, indent=2)

    return None

def count_keypairs(data):
    """Count keypairs in keystore data"""
    try:
        if 'keystore' in data and 'keypairs' in data['keystore']:
            return len(data['keystore']['keypairs'])
        return 0
    except:
        return 0
